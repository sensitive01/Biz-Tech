1. AUTHENTICATION & HEADERS
================================================================================
Most protected endpoints require an Authorization header containing the JWT token
obtained from the login or registration response:

Headers:
  Content-Type: application/json
  Authorization: Bearer <JWT_TOKEN>

For multipart file uploads (documents, receipts, payment proofs):
  Headers:
    Authorization: Bearer <JWT_TOKEN>  (if endpoint requires auth)
  Body: multipart/form-data

Note on Login:
The login endpoint supports logging in with EITHER:
  1) Email address + Password
  2) 10-digit Phone number + Password

[POST] /api/auth/register
Desc: Register a new workspace owner account
Auth: None
Content-Type: application/json
Request Body:
{
  "fullName": "John Doe",
  "email": "owner@example.com",
  "phone": "9876543210",       // Exactly 10 digits, must be globally unique
  "password": "SecurePassword123",
  "businessType": 1            // 1: Sole Operator, 2: Single Branch + Staff, 3: Multi-Branch, 4: Multi-Business
}
Response (200 OK):
{
  "token": "eyJhbGciOiJIUzI1NiIsIn...",

    "id": "67401a...",

    "email": "owner@example.com",
    "phone": "9876543210",
    "businessType": 1,
    "modules": ["sales", "expenses", "customers", "documents", "reminders"]
  }
}

import 'dart:convert';
import 'package:http/http.dart' as http;

class BizTechApi {
  // Use 10.0.2.2 for Android Emulator, localhost for iOS simulator
  static const String baseUrl = 'http://10.0.2.2:5000/api';

  // 1. Universal Login (Supports Phone OR Email)
  static Future<Map<String, dynamic>> login({
    required String identifier, // Phone number or Email
    required String password,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/auth/login'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'email': identifier, // The backend checks both email and phone automatically
        'password': password,
      }),
    );
    final data = jsonDecode(response.body);
    if (response.statusCode == 200) {
      return data; // contains 'token' and 'user'
    } else {
      throw Exception(data['message'] ?? 'Login failed');
    }
  }

  // 2. Authenticated GET Request
  static Future<List<dynamic>> getProducts(String token) async {
    final response = await http.get(
      Uri.parse('$baseUrl/products'),
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $token',
      },
    );
    if (response.statusCode == 200) {
      return jsonDecode(response.body);
    } else {
      throw Exception('Failed to load products');
    }
  }

  // 3. Upload Document to Cloudinary
  static Future<String> uploadDocument({
    required String token,
    required String filePath,
  }) async {
    var request = http.MultipartRequest(
      'POST',
      Uri.parse('$baseUrl/documents/upload'),
    );
    request.headers['Authorization'] = 'Bearer $token';
    request.files.add(await http.MultipartFile.fromPath('file', filePath));

    var streamedResponse = await request.send();
    var response = await http.Response.fromStream(streamedResponse);
    final data = jsonDecode(response.body);

    if (data['success'] == true) {
      return data['fileUrl']; // Direct Cloudinary CDN URL
    } else {
      throw Exception(data['message'] ?? 'Upload failed');
    }
  }
}
